/// Cliente HTTP único (Singleton) configurado con Hive Cache y reintentos automáticos para Error 429
class ApiClient {
  static final ApiClient _instance = ApiClient._internal();
  late final Dio dio;

  factory ApiClient() => _instance;

  ApiClient._internal() {
    dio = Dio(
      BaseOptions(
        persistentConnection: true,
        connectTimeout: const Duration(seconds: 20),
        receiveTimeout: const Duration(seconds: 20),
      ),
    );

    // 1. Configuración de Caché Persistente con Hive
    final cacheOptions = CacheOptions(
      store: HiveCacheStore(
        Hive.box('api_cache').path,
      ), // Usa tu directorio/caja existente de Hive
      policy: CachePolicy
          .refreshForceCache, // Busca en caché primero; si expira, descarga
      maxStale: const Duration(days: 1), // Duración de los datos en caché
      priority: CachePriority.normal,
    );

    dio.interceptors.add(DioCacheInterceptor(options: cacheOptions));

    // 2. Interceptor para manejar el error 429 (Too Many Requests)
    dio.interceptors.add(
      InterceptorsWrapper(
        onError: (DioException err, ErrorInterceptorHandler handler) async {
          if (err.response?.statusCode == 429) {
            // Lee el delay enviado por tu servidor Laravel o usa 2 segundos por defecto
            final retryAfter =
                int.tryParse(
                  err.response?.headers.value('retry-after') ?? '',
                ) ??
                2;
            await Future.delayed(Duration(seconds: retryAfter));

            try {
              final response = await dio.fetch(err.requestOptions);
              return handler.resolve(response);
            } catch (e) {
              return handler.next(err);
            }
          }
          return handler.next(err);
        },
      ),
    );
  }
}

  static String sanitizeToFriendlyUrl(String input) {
    // Convertir a minúsculas
    String sanitized = input.toLowerCase();

    // Reemplazar caracteres acentuados o especiales por sus equivalentes simples
    sanitized = sanitized
        .replaceAll(RegExp(r'[áàäâã]'), 'a')
        .replaceAll(RegExp(r'[éèëê]'), 'e')
        .replaceAll(RegExp(r'[íìïî]'), 'i')
        .replaceAll(RegExp(r'[óòöôõ]'), 'o')
        .replaceAll(RegExp(r'[úùüû]'), 'u')
        .replaceAll(RegExp(r'ñ'), 'n')
        .replaceAll(RegExp(r'ç'), 'c');

    // Reemplazar cualquier carácter no alfanumérico o espacios con un guion
    sanitized = sanitized.replaceAll(RegExp(r'[^a-z0-9]+'), '-');

    // Eliminar guiones adicionales al inicio o final
    sanitized = sanitized.replaceAll(RegExp(r'^-+|-+$'), '');

    return sanitized;
  }

  /// Método privado para limpiar y construir las URLs según plataforma y entorno
  static String _buildUrl(String endpoint) {
    late String stringUrl = "";
    if (Platform.isAndroid) {
      if (isDebug) {
        stringUrl = isDevice
            ? "http://192.168.0"
            : "https://app.ahorrala.com/public/api";
      }
      stringUrl = "https://app.ahorrala.com/public/api";
    } else if (Platform.isIOS) {
      if (isDebug) {
        stringUrl = isDevice
            ? "http://192.168.0"
            : "http://app.ahorrala.com/public/api/";
      }
      stringUrl = "https://app.ahorrala.com/public/api/";
    }
    stringUrl = "https://app.ahorrala.com/public/api/";
    stringUrl = stringUrl + endpoint;
    return stringUrl;
  }

  static Future<String?> getDeviceUniqueIdentifier() async {
    final DeviceInfoPlugin deviceInfo = DeviceInfoPlugin();

    if (Platform.isAndroid) {
      final AndroidDeviceInfo androidInfo = await deviceInfo.androidInfo;
      // Returns the hardware-based build ID string unique to the device configuration
      return androidInfo.id;
    } else if (Platform.isIOS) {
      final IosDeviceInfo iosInfo = await deviceInfo.iosInfo;
      // Returns the vendor-specific unique device token
      return iosInfo.identifierForVendor;
    }

    return null;
  }

  static Future<Map<String, String>> obtenerDatosControl() async {
    final DeviceInfoPlugin deviceInfo = DeviceInfoPlugin();
    String idUnico = '';
    String plataforma = '';

    try {
      if (Platform.isAndroid) {
        AndroidDeviceInfo androidInfo = await deviceInfo.androidInfo;
        idUnico = androidInfo.id; // ID único del hardware en Android
        plataforma = 'android';
      } else if (Platform.isIOS) {
        IosDeviceInfo iosInfo = await deviceInfo.iosInfo;
        // ID único para tu app en iOS (cambia si el usuario desinstala todas tus apps)
        idUnico = iosInfo.identifierForVendor ?? '';
        plataforma = 'ios';
      }
    } catch (e) {
      idUnico = 'error_id';
    }

    return {'deviceId': idUnico, 'plataforma': plataforma};
  }
